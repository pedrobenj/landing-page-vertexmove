# Loja Vertex — Etapa 1: Plano de Implementação

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Entregar a loja de camisas e medalhas (catálogo, produto, carrinho, checkout até a criação do pedido, página do pedido e admin de produtos) funcionando com dados mockados, sem banco e sem Pix.

**Architecture:** Um `StoreRepository` (interface) com duas implementações — mock em memória (dev/testes, ativada por `USE_MOCK_DATA`) e Prisma (produção). Regras de negócio puras em `src/server/services/` (`priceCart`, `createOrder`) recebem o repositório por parâmetro, o que permite testá-las com Vitest sem banco. O carrinho vive em Zustand persistido em `localStorage` e guarda só `variantId` + `quantity`; todo preço e estoque vem do servidor via server actions.

**Tech Stack:** Next.js 16 (App Router, Cache Components), Prisma 7, Zustand, TanStack Query, React Hook Form + Zod 4, shadcn/ui, Vitest (novo, para regras puras), Playwright (e2e).

**Referências obrigatórias:**
- `CLAUDE.md` (convenções: um componente por arquivo, item de `.map()` em componente próprio, helpers em `src/lib/utils/`, 4 estados de dados, `HttpResponse<T>`).
- `docs/plans/2026-11-04-loja-design.md` (desenho validado).
- Next 16: docs locais em `node_modules/next/dist/docs/`. Com Cache Components, qualquer leitura de dados ou `params` precisa estar dentro de `<Suspense>` e/ou após `await connection()`.

**Desvios conscientes em relação ao desenho:**
1. O catálogo `/loja` mostra duas seções ("Camisas" e "Medalhas") com âncoras em vez de um filtro por search param — o `CLAUDE.md` proíbe filtros em search params e o catálogo é pequeno. `/medalhas` redireciona para `/loja#medalhas`.
2. `ProductVariant` ganha `position` para manter a ordem P, M, G, GG.
3. Fotos de produto ficam para quando o storage (Blob/R2) for definido; até lá o card mostra um placeholder da marca. O admin não tem upload nesta etapa.
4. Na edição de produto, tamanhos já salvos não podem ser removidos (só editados ou zerados), para não quebrar o histórico de pedidos.
5. Login do admin em modo mock: `admin@vertex.dev` / `vertex-admin-123` (só com `USE_MOCK_DATA=true`, que já é bloqueado em produção).

**Fora desta etapa (Etapa 2):** Pix, webhook, expiração automática, notificações, admin de pedidos, card da loja no dashboard, `CHECK` de `Payment` na migration (ver Task 2, passo 4), consulta de pedidos em `/consulta`.

```ts
// src/lib/utils/generate-code.ts
import { randomInt } from "node:crypto"

// Sem 0/O, 1/I/L para leitura fácil no WhatsApp e na retirada.
const ALPHABET = "23456789ABCDEFGHJKMNPQRSTUVWXYZ"

/** Somente servidor (usa node:crypto). */
function generateCode(prefix: string, length = 4): string {
  let code = ""
  for (let i = 0; i < length; i++) {
    code += ALPHABET[randomInt(ALPHABET.length)]
  }
  return `${prefix}-${code}`
}

/** Inscrição, ex: VTX-7K3P */
export function generateRegistrationCode(): string {
  return generateCode("VTX")
}

/** Pedido da loja, ex: VTX-P-7K3P */
export function generateOrderCode(): string {
  return generateCode("VTX-P")
}
```

```ts
// src/lib/utils/currency-input.ts
/** Converte "1.234,56" / "R$ 89,90" / "89" em centavos. Retorna null se inválido ou negativo. */
export function parseCurrencyInput(value: string): number | null {
  const cleaned = value.replace(/R\$\s?/, "").trim().replace(/\./g, "")
  if (!/^\d+(,\d{1,2})?$/.test(cleaned)) return null
  const [reais, cents = ""] = cleaned.split(",")
  return Number(reais) * 100 + Number(cents.padEnd(2, "0"))
}

/** Centavos → texto do input ("89,90"). */
export function formatCentsInput(cents: number): string {
  return (cents / 100).toFixed(2).replace(".", ",")
}
```

```ts
export function slugify(value: string): string {
  return value
    .normalize("NFD")
    .replace(/[̀-ͯ]/g, "")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
}
```

```ts
// src/server/services/cart.service.ts
import "server-only"

import { CART_ISSUE_REASON } from "@/enums/cart-issue-reason.enum"
import type { CartIssue, CartItem, PricedCart, PricedCartLine } from "@/types/cart.interface"
import type { StoreProduct, StoreVariant } from "@/types/store-product.interface"

interface VariantMatch {
  product: StoreProduct
  variant: StoreVariant
}

function indexVariants(products: StoreProduct[]): Map<string, VariantMatch> {
  const index = new Map<string, VariantMatch>()
  for (const product of products) {
    for (const variant of product.variants) {
      index.set(variant.id, { product, variant })
    }
  }
  return index
}

function describeItem({ product, variant }: VariantMatch): string {
  return `${product.name} (${variant.label})`
}

/**
 * Precifica o carrinho com os dados ATUAIS do servidor (preço, estoque, produto ativo).
 * Nunca usa valores vindos do navegador além de variantId e quantity.
 */
export function priceCart(items: CartItem[], products: StoreProduct[]): PricedCart {
  const index = indexVariants(products)
  const lines: PricedCartLine[] = []
  const issues: CartIssue[] = []

  for (const item of items) {
    const match = index.get(item.variantId)

    if (!match || !match.product.isActive) {
      issues.push({
        variantId: item.variantId,
        reason: CART_ISSUE_REASON.UNAVAILABLE,
        availableQuantity: 0,
        message: "Um item do seu carrinho não está mais disponível",
      })
      continue
    }

    if (match.variant.stock <= 0) {
      issues.push({
        variantId: item.variantId,
        reason: CART_ISSUE_REASON.INSUFFICIENT_STOCK,
        availableQuantity: 0,
        message: `${describeItem(match)} esgotou`,
      })
      continue
    }

    const quantity = Math.min(item.quantity, match.variant.stock)
    if (quantity < item.quantity) {
      issues.push({
        variantId: item.variantId,
        reason: CART_ISSUE_REASON.INSUFFICIENT_STOCK,
        availableQuantity: quantity,
        message: `Só restam ${quantity} unidades de ${describeItem(match)}`,
      })
    }

    lines.push({
      variantId: match.variant.id,
      productId: match.product.id,
      productSlug: match.product.slug,
      productName: match.product.name,
      productType: match.product.type,
      variantLabel: match.variant.label,
      imageUrl: match.product.imageUrls[0] ?? null,
      unitPrice: match.product.price,
      quantity,
      lineTotal: match.product.price * quantity,
      pickupInfo: match.product.pickupInfo,
    })
  }

  return {
    lines,
    issues,
    total: lines.reduce((sum, line) => sum + line.lineTotal, 0),
    itemCount: lines.reduce((sum, line) => sum + line.quantity, 0),
  }
}
```

```ts
// src/server/repositories/mock-store.repository.ts
import { randomUUID } from "node:crypto"

import { generateOrderCode } from "@/lib/utils/generate-code"
import { getFirstName } from "@/lib/utils/get-first-name"
import { DuplicateSlugError } from "@/server/errors"
import type { ProductFormData } from "@/lib/validations/product.schema"
import type { NewOrderData, OrderSummary } from "@/types/order.interface"
import type { StoreProduct } from "@/types/store-product.interface"

import type { StoreRepository } from "./store-repository.interface"

interface MockOrder extends OrderSummary {
  buyerCpf: string
  variantIds: string[]
}

/** Repositório em memória para desenvolvimento e testes. Estado some ao reiniciar o servidor. */
export function createMockStoreRepository(seed: StoreProduct[]): StoreRepository {
  const products: StoreProduct[] = structuredClone(seed)
  const orders = new Map<string, MockOrder>()

  const clone = <T>(value: T): T => structuredClone(value)
  const findVariant = (variantId: string) => {
    for (const product of products) {
      const variant = product.variants.find((v) => v.id === variantId)
      if (variant) return { product, variant }
    }
    return null
  }

  return {
    async listActiveProducts() {
      return clone(products.filter((p) => p.isActive))
    },

    async listFeaturedProducts(limit) {
      return clone(products.filter((p) => p.isActive && p.isFeatured).slice(0, limit))
    },

    async getActiveProductBySlug(slug) {
      const product = products.find((p) => p.slug === slug && p.isActive)
      return product ? clone(product) : null
    },

    async getProductsByVariantIds(variantIds) {
      const ids = new Set(variantIds)
      return clone(products.filter((p) => p.variants.some((v) => ids.has(v.id))))
    },

    async createPendingOrder(data: NewOrderData) {
      // Verifica tudo antes de baixar (equivale à transação no Prisma).
      const outOfStock = data.lines
        .filter((line) => (findVariant(line.variantId)?.variant.stock ?? 0) < line.quantity)
        .map((line) => line.variantId)
      if (outOfStock.length > 0) return { ok: false, outOfStockVariantIds: outOfStock }

      for (const line of data.lines) {
        const match = findVariant(line.variantId)
        if (match) match.variant.stock -= line.quantity
      }

      const pickupInfos = [...new Set(data.lines.map((l) => findVariant(l.variantId)?.product.pickupInfo ?? ""))].filter(Boolean)
      const code = generateOrderCode()
      orders.set(code, {
        code,
        status: "PENDING",
        amount: data.amount,
        buyerFirstName: getFirstName(data.buyer.name),
        buyerCpf: data.buyer.cpf,
        createdAt: new Date(),
        expiresAt: data.expiresAt,
        items: data.lines.map(({ productName, variantLabel, quantity, unitPrice }) => ({ productName, variantLabel, quantity, unitPrice })),
        pickupInfos,
        variantIds: data.lines.map((l) => l.variantId),
      })
      return { ok: true, code }
    },

    async getOrderSummaryByCode(code) {
      const found = orders.get(code)
      if (!found) return null
      const { buyerCpf: _cpf, variantIds: _ids, ...summary } = found
      return clone(summary)
    },

    async listAllProducts() {
      return clone(products)
    },

    async getProductById(id) {
      const product = products.find((p) => p.id === id)
      return product ? clone(product) : null
    },

    async saveProduct(data: ProductFormData, id?: string) {
      if (products.some((p) => p.slug === data.slug && p.id !== id)) throw new DuplicateSlugError()

      const { variants, ...fields } = data
      const existing = id ? products.find((p) => p.id === id) : undefined

      if (existing) {
        Object.assign(existing, fields)
        existing.variants = variants.map((variant) => ({
          id: variant.id ?? `mock-var-${randomUUID()}`,
          label: variant.label,
          stock: variant.stock,
        }))
        return clone(existing)
      }

      const created: StoreProduct = {
        ...fields,
        id: `mock-prod-${randomUUID()}`,
        imageUrls: [],
        variants: variants.map((variant) => ({ id: `mock-var-${randomUUID()}`, label: variant.label, stock: variant.stock })),
      }
      products.push(created)
      return clone(created)
    },
  }
}
```

```ts
// src/server/repositories/prisma-store.repository.ts
import "server-only"

import { Prisma } from "@/generated/prisma/client"
import { OrderStatus } from "@/generated/prisma/enums"
import { generateOrderCode } from "@/lib/utils/generate-code"
import { getFirstName } from "@/lib/utils/get-first-name"
import type { ProductFormData } from "@/lib/validations/product.schema"
import { db } from "@/server/db"
import { DuplicateSlugError } from "@/server/errors"
import type { StoreProduct } from "@/types/store-product.interface"

import type { StoreRepository } from "./store-repository.interface"

const productInclude = {
  variants: { orderBy: { position: "asc" }, select: { id: true, label: true, stock: true } },
} satisfies Prisma.ProductInclude

type ProductWithVariants = Prisma.ProductGetPayload<{ include: typeof productInclude }>

function toStoreProduct(product: ProductWithVariants): StoreProduct {
  return {
    id: product.id,
    slug: product.slug,
    name: product.name,
    description: product.description,
    type: product.type,
    imageUrls: product.imageUrls,
    price: product.price,
    pickupInfo: product.pickupInfo,
    isActive: product.isActive,
    isFeatured: product.isFeatured,
    variants: product.variants,
  }
}

function isUniqueViolation(error: unknown): boolean {
  return error instanceof Prisma.PrismaClientKnownRequestError && error.code === "P2002"
}

class OutOfStockError extends Error {
  constructor(readonly variantIds: string[]) {
    super("Estoque insuficiente")
  }
}

const MAX_CODE_ATTEMPTS = 3

export const prismaStoreRepository: StoreRepository = {
  async listActiveProducts() {
    const products = await db.product.findMany({
      where: { isActive: true },
      orderBy: [{ type: "asc" }, { createdAt: "desc" }],
      include: productInclude,
    })
    return products.map(toStoreProduct)
  },

  async listFeaturedProducts(limit) {
    const products = await db.product.findMany({
      where: { isActive: true, isFeatured: true },
      orderBy: { createdAt: "desc" },
      take: limit,
      include: productInclude,
    })
    return products.map(toStoreProduct)
  },

  async getActiveProductBySlug(slug) {
    const product = await db.product.findFirst({ where: { slug, isActive: true }, include: productInclude })
    return product ? toStoreProduct(product) : null
  },

  async getProductsByVariantIds(variantIds) {
    const products = await db.product.findMany({
      where: { variants: { some: { id: { in: variantIds } } } },
      include: productInclude,
    })
    return products.map(toStoreProduct)
  },

  async createPendingOrder(data) {
    for (let attempt = 1; attempt <= MAX_CODE_ATTEMPTS; attempt++) {
      try {
        return await db.$transaction(async (tx) => {
          const outOfStock: string[] = []
          for (const line of data.lines) {
            // Update condicional: só baixa se ainda houver estoque. Evita vender além do que existe.
            const { count } = await tx.productVariant.updateMany({
              where: { id: line.variantId, stock: { gte: line.quantity } },
              data: { stock: { decrement: line.quantity } },
            })
            if (count === 0) outOfStock.push(line.variantId)
          }
          if (outOfStock.length > 0) throw new OutOfStockError(outOfStock)

          const order = await tx.order.create({
            data: {
              code: generateOrderCode(),
              buyerName: data.buyer.name,
              buyerCpf: data.buyer.cpf,
              buyerWhatsapp: data.buyer.whatsapp,
              buyerEmail: data.buyer.email,
              amount: data.amount,
              status: OrderStatus.PENDING,
              expiresAt: data.expiresAt,
              privacyAcceptedAt: data.privacyAcceptedAt,
              items: { create: data.lines },
            },
            select: { code: true },
          })
          return { ok: true as const, code: order.code }
        })
      } catch (error) {
        if (error instanceof OutOfStockError) return { ok: false, outOfStockVariantIds: error.variantIds }
        if (isUniqueViolation(error) && attempt < MAX_CODE_ATTEMPTS) continue // colisão de código: tenta outro
        throw error
      }
    }
    throw new Error("Não foi possível gerar um código de pedido único")
  },

  async getOrderSummaryByCode(code) {
    const order = await db.order.findUnique({
      where: { code },
      include: { items: { include: { variant: { select: { product: { select: { pickupInfo: true } } } } } } },
    })
    if (!order) return null
    return {
      code: order.code,
      status: order.status,
      amount: order.amount,
      buyerFirstName: getFirstName(order.buyerName),
      createdAt: order.createdAt,
      expiresAt: order.expiresAt,
      items: order.items.map(({ productName, variantLabel, quantity, unitPrice }) => ({ productName, variantLabel, quantity, unitPrice })),
      pickupInfos: [...new Set(order.items.map((item) => item.variant.product.pickupInfo))],
    }
  },

  async listAllProducts() {
    const products = await db.product.findMany({ orderBy: { createdAt: "desc" }, include: productInclude })
    return products.map(toStoreProduct)
  },

  async getProductById(id) {
    const product = await db.product.findUnique({ where: { id }, include: productInclude })
    return product ? toStoreProduct(product) : null
  },

  async saveProduct(data: ProductFormData, id?: string) {
    const { variants, ...fields } = data
    try {
      if (!id) {
        const created = await db.product.create({
          data: {
            ...fields,
            imageUrls: [],
            variants: { create: variants.map((v, position) => ({ label: v.label, stock: v.stock, position })) },
          },
          include: productInclude,
        })
        return toStoreProduct(created)
      }

      const updated = await db.$transaction(async (tx) => {
        await tx.product.update({ where: { id }, data: fields })
        for (const [position, variant] of variants.entries()) {
          if (variant.id) {
            await tx.productVariant.update({ where: { id: variant.id }, data: { label: variant.label, stock: variant.stock, position } })
          } else {
            await tx.productVariant.create({ data: { productId: id, label: variant.label, stock: variant.stock, position } })
          }
        }
        return tx.product.findUniqueOrThrow({ where: { id }, include: productInclude })
      })
      return toStoreProduct(updated)
    } catch (error) {
      if (isUniqueViolation(error)) throw new DuplicateSlugError()
      throw error
    }
  },
}
```

```ts
// src/server/services/order.service.ts
import "server-only"

import { addMinutes } from "date-fns"

import { ORDER_RESERVATION_MINUTES } from "@/lib/utils/constants"
import type { CheckoutData } from "@/lib/validations/checkout.schema"
import type { StoreRepository } from "@/server/repositories/store-repository.interface"
import type { PricedCart } from "@/types/cart.interface"

import { priceCart } from "./cart.service"

export type CreateOrderResult =
  | { ok: true; code: string }
  | { ok: false; reason: "EMPTY_CART" }
  | { ok: false; reason: "CART_CHANGED"; cart: PricedCart }

async function priceCurrentCart(input: CheckoutData, repo: StoreRepository): Promise<PricedCart> {
  const products = await repo.getProductsByVariantIds(input.items.map((item) => item.variantId))
  return priceCart(input.items, products)
}

export async function createOrder(input: CheckoutData, repo: StoreRepository, now = new Date()): Promise<CreateOrderResult> {
  const cart = await priceCurrentCart(input, repo)
  if (cart.issues.length > 0) return { ok: false, reason: "CART_CHANGED", cart }
  if (cart.lines.length === 0) return { ok: false, reason: "EMPTY_CART" }

  const result = await repo.createPendingOrder({
    buyer: { name: input.name, cpf: input.cpf, whatsapp: input.whatsapp, email: input.email },
    amount: cart.total,
    expiresAt: addMinutes(now, ORDER_RESERVATION_MINUTES),
    privacyAcceptedAt: now,
    lines: cart.lines.map(({ variantId, quantity, unitPrice, productName, variantLabel }) => ({
      variantId,
      quantity,
      unitPrice,
      productName,
      variantLabel,
    })),
  })

  if (result.ok) return { ok: true, code: result.code }
  // Outra compra levou o estoque entre a validação e a baixa: devolve o carrinho atualizado.
  return { ok: false, reason: "CART_CHANGED", cart: await priceCurrentCart(input, repo) }
}
```

```ts
// src/server/rate-limit.ts
/**
 * Limite simples por janela fixa, em memória (por instância do servidor).
 * Suficiente para o MVP; em produção com várias instâncias, trocar por Upstash/Redis.
 */
export function createRateLimiter() {
  const hits = new Map<string, { count: number; resetAt: number }>()

  return function isLimited(key: string, limit: number, windowMs: number, now = Date.now()): boolean {
    const entry = hits.get(key)
    if (!entry || entry.resetAt <= now) {
      hits.set(key, { count: 1, resetAt: now + windowMs })
      return false
    }
    entry.count += 1
    return entry.count > limit
  }
}
```

```ts
// src/server/request-rate-limit.ts
import "server-only"

import { headers } from "next/headers"

import { createRateLimiter } from "./rate-limit"

const globalForRateLimit = globalThis as unknown as { isLimited?: ReturnType<typeof createRateLimiter> }
const isLimited = (globalForRateLimit.isLimited ??= createRateLimiter())

/** Rate limit por IP para server actions. */
export async function isRequestRateLimited(scope: string, limit: number, windowMs: number): Promise<boolean> {
  const forwardedFor = (await headers()).get("x-forwarded-for")
  const ip = forwardedFor?.split(",")[0]?.trim() || "unknown"
  return isLimited(`${scope}:${ip}`, limit, windowMs)
}
```

```ts
"use server"

import { revalidatePath } from "next/cache"

import { cartItemsSchema } from "@/lib/validations/cart.schema"
import { checkoutSchema, type CheckoutInput } from "@/lib/validations/checkout.schema"
import { getStoreRepository } from "@/server/repositories/store.repository"
import { isRequestRateLimited } from "@/server/request-rate-limit"
import { priceCart } from "@/server/services/cart.service"
import { createOrder } from "@/server/services/order.service"
import type { CartIssue, CartItem, PricedCart } from "@/types/cart.interface"
import type { HttpResponse } from "@/types/http-response.interface"

const CHECKOUT_LIMIT = 10
const CHECKOUT_WINDOW_MS = 60_000

export async function validateCartAction(items: CartItem[]): Promise<HttpResponse<PricedCart>> {
  const parsed = cartItemsSchema.safeParse(items)
  if (!parsed.success) {
    return { status: 400, error: "INVALID_CART", title: parsed.error.issues[0]?.message ?? "Carrinho inválido" }
  }

  try {
    const repo = getStoreRepository()
    const products = await repo.getProductsByVariantIds(parsed.data.map((item) => item.variantId))
    return { status: 200, data: priceCart(parsed.data, products) }
  } catch (error) {
    console.error("[validateCartAction]", error)
    return { status: 500, error: "INTERNAL", title: "Não foi possível carregar o carrinho" }
  }
}

export interface CheckoutResponseData {
  code?: string
  issues?: CartIssue[]
}

export async function createOrderAction(input: CheckoutInput): Promise<HttpResponse<CheckoutResponseData>> {
  if (await isRequestRateLimited("checkout", CHECKOUT_LIMIT, CHECKOUT_WINDOW_MS)) {
    return { status: 429, error: "RATE_LIMITED", title: "Muitas tentativas. Aguarde um minuto e tente de novo." }
  }

  const parsed = checkoutSchema.safeParse(input)
  if (!parsed.success) {
    return { status: 400, error: "INVALID_INPUT", title: parsed.error.issues[0]?.message ?? "Verifique seus dados" }
  }

  try {
    const result = await createOrder(parsed.data, getStoreRepository())

    if (result.ok) {
      revalidatePath("/loja")
      return { status: 201, data: { code: result.code }, title: "Pedido criado!" }
    }
    if (result.reason === "EMPTY_CART") {
      return { status: 400, error: "EMPTY_CART", title: "Seu carrinho está vazio" }
    }
    return {
      status: 409,
      error: "CART_CHANGED",
      title: result.cart.issues[0]?.message ?? "Seu carrinho mudou. Confira os itens.",
      data: { issues: result.cart.issues },
    }
  } catch (error) {
    console.error("[createOrderAction]", error)
    return { status: 500, error: "INTERNAL", title: "Não foi possível criar o pedido. Tente novamente." }
  }
}
```

```ts
// src/stores/cart-store.ts
import { create } from "zustand"
import { createJSONStorage, persist } from "zustand/middleware"

import { MAX_ITEMS_PER_ORDER, MAX_QUANTITY_PER_ITEM } from "@/lib/utils/constants"
import type { CartIssue, CartItem } from "@/types/cart.interface"

interface CartState {
  items: CartItem[]
  /** Retorna false se o carrinho já tem o máximo de itens diferentes. */
  addItem: (variantId: string, quantity: number) => boolean
  setQuantity: (variantId: string, quantity: number) => void
  removeItem: (variantId: string) => void
  applyIssues: (issues: CartIssue[]) => void
  clear: () => void
}

const clampQuantity = (quantity: number) => Math.max(1, Math.min(quantity, MAX_QUANTITY_PER_ITEM))

/**
 * Carrinho da loja. Guarda só variantId + quantity; preço e estoque vêm sempre do servidor.
 * skipHydration: a leitura do localStorage acontece no CartHydrator, depois da hidratação do React.
 */
export const useCartStore = create<CartState>()(
  persist(
    (set, get) => ({
      items: [],
      addItem: (variantId, quantity) => {
        const items = get().items
        const existing = items.find((item) => item.variantId === variantId)
        if (existing) {
          set({
            items: items.map((item) =>
              item.variantId === variantId ? { ...item, quantity: clampQuantity(item.quantity + quantity) } : item,
            ),
          })
          return true
        }
        if (items.length >= MAX_ITEMS_PER_ORDER) return false
        set({ items: [...items, { variantId, quantity: clampQuantity(quantity) }] })
        return true
      },
      setQuantity: (variantId, quantity) =>
        set({
          items: get().items.map((item) =>
            item.variantId === variantId ? { ...item, quantity: clampQuantity(quantity) } : item,
          ),
        }),
      removeItem: (variantId) => set({ items: get().items.filter((item) => item.variantId !== variantId) }),
      applyIssues: (issues) => {
        const byVariant = new Map(issues.map((issue) => [issue.variantId, issue.availableQuantity]))
        set({
          items: get()
            .items.map((item) =>
              byVariant.has(item.variantId) ? { ...item, quantity: byVariant.get(item.variantId) ?? 0 } : item,
            )
            .filter((item) => item.quantity > 0),
        })
      },
      clear: () => set({ items: [] }),
    }),
    {
      name: "vertex-cart",
      storage: createJSONStorage(() => localStorage),
      partialize: (state) => ({ items: state.items }),
      skipHydration: true,
    },
  ),
)

export const selectCartCount = (state: CartState): number =>
  state.items.reduce((total, item) => total + item.quantity, 0)
```

```tsx
// src/app/(public)/loja/[slug]/components/product-purchase-form.tsx
"use client"

import { useRouter } from "next/navigation"
import { useState } from "react"
import { toast } from "sonner"

import { QuantityStepper } from "@/components/store"
import { Button } from "@/components/ui/button"
import { RadioGroup } from "@/components/ui/radio-group"
import { MAX_QUANTITY_PER_ITEM } from "@/lib/utils/constants"
import { useCartStore } from "@/stores/cart-store"
import type { StoreVariant } from "@/types/store-product.interface"

import { SizeOption } from "./size-option"

interface ProductPurchaseFormProps {
  variants: StoreVariant[]
}

export function ProductPurchaseForm({ variants }: ProductPurchaseFormProps) {
  const router = useRouter()
  const addItem = useCartStore((state) => state.addItem)
  const firstAvailable = variants.find((variant) => variant.stock > 0)
  const [variantId, setVariantId] = useState(firstAvailable?.id ?? "")
  const [quantity, setQuantity] = useState(1)

  const selected = variants.find((variant) => variant.id === variantId)
  const maxQuantity = Math.min(selected?.stock ?? 1, MAX_QUANTITY_PER_ITEM)
  const hasSizeChoice = variants.length > 1

  if (!firstAvailable) {
    return (
      <Button size="xl" disabled className="w-full sm:w-auto">
        Esgotado
      </Button>
    )
  }

  function handleAdd() {
    if (!selected) return
    const added = addItem(selected.id, quantity)
    if (!added) {
      toast.error("Seu carrinho já tem o máximo de itens diferentes")
      return
    }
    toast.success("Adicionado ao carrinho", {
      action: { label: "Ver carrinho", onClick: () => router.push("/loja/carrinho") },
    })
  }

  return (
    <div className="flex flex-col gap-6">
      {hasSizeChoice && (
        <fieldset className="flex flex-col gap-2">
          <legend className="mb-2 text-sm font-semibold">Tamanho</legend>
          <RadioGroup
            value={variantId}
            onValueChange={(value) => {
              setVariantId(value)
              setQuantity(1)
            }}
            className="flex flex-wrap gap-2"
          >
            {variants.map((variant) => (
              <SizeOption key={variant.id} variant={variant} />
            ))}
          </RadioGroup>
        </fieldset>
      )}

      <div className="flex flex-col gap-2">
        <span className="text-sm font-semibold">Quantidade</span>
        <QuantityStepper value={quantity} max={maxQuantity} onChange={setQuantity} label="Quantidade" />
      </div>

      <Button variant="cta" size="xl" className="w-full sm:w-auto" onClick={handleAdd}>
        Adicionar ao carrinho
      </Button>
    </div>
  )
}
```

```ts
// src/hooks/data/use-priced-cart.ts
"use client"

import { useQuery } from "@tanstack/react-query"
import { useEffect } from "react"
import { toast } from "sonner"

import { validateCartAction } from "@/app/actions/store"
import { useCartHydrated } from "@/hooks/use-cart-hydrated"
import { queryKeys } from "@/lib/utils"
import { useCartStore } from "@/stores/cart-store"

/** Carrinho precificado pelo servidor. Aplica automaticamente os avisos (esgotado, estoque menor). */
export function usePricedCart() {
  const hydrated = useCartHydrated()
  const items = useCartStore((state) => state.items)
  const applyIssues = useCartStore((state) => state.applyIssues)

  const query = useQuery({
    queryKey: queryKeys.store.cart(items),
    queryFn: async () => {
      const response = await validateCartAction(items)
      if (!response.data) throw new Error(response.title ?? "Não foi possível carregar o carrinho")
      return response.data
    },
    enabled: hydrated && items.length > 0,
    staleTime: 0,
  })

  const issues = query.data?.issues
  useEffect(() => {
    if (!issues || issues.length === 0) return
    for (const issue of issues) toast.warning(issue.message)
    applyIssues(issues)
  }, [issues, applyIssues])

  return { ...query, hydrated, isEmpty: hydrated && items.length === 0 }
}
```

```tsx
"use client"

import { zodResolver } from "@hookform/resolvers/zod"
import Link from "next/link"
import { useRouter } from "next/navigation"
import { useTransition } from "react"
import { Controller, useForm } from "react-hook-form"
import { toast } from "sonner"

import { createOrderAction } from "@/app/actions/store"
import { Button } from "@/components/ui/button"
import { Checkbox } from "@/components/ui/checkbox"
import { Field, FieldError, FieldGroup, FieldLabel } from "@/components/ui/field"
import { Input } from "@/components/ui/input"
import { formatCpfInput, formatPhoneInput } from "@/lib/utils"
import { checkoutFormSchema, type CheckoutFormInput } from "@/lib/validations/checkout.schema"
import { useCartStore } from "@/stores/cart-store"

export function CheckoutForm() {
  const router = useRouter()
  const [isPending, startTransition] = useTransition()
  const items = useCartStore((state) => state.items)
  const clear = useCartStore((state) => state.clear)
  const applyIssues = useCartStore((state) => state.applyIssues)

  const form = useForm<CheckoutFormInput>({
    resolver: zodResolver(checkoutFormSchema),
    defaultValues: { name: "", cpf: "", whatsapp: "", email: "", acceptPrivacy: false as unknown as true },
  })

  function onSubmit(values: CheckoutFormInput) {
    startTransition(async () => {
      const response = await createOrderAction({ ...values, items })

      if (response.status === 201 && response.data?.code) {
        clear()
        router.push(`/pedido/${response.data.code}`)
        return
      }
      if (response.status === 409 && response.data?.issues) {
        applyIssues(response.data.issues)
        toast.error(response.title ?? "Seu carrinho mudou")
        router.push("/loja/carrinho")
        return
      }
      toast.error(response.title ?? "Não foi possível finalizar a compra")
    })
  }

  return (
    <form onSubmit={form.handleSubmit(onSubmit)} noValidate aria-label="Dados do comprador">
      <FieldGroup>
        <Controller
          name="name"
          control={form.control}
          render={({ field, fieldState }) => (
            <Field data-invalid={fieldState.invalid}>
              <FieldLabel htmlFor="buyer-name">Nome completo</FieldLabel>
              <Input {...field} id="buyer-name" autoComplete="name" aria-invalid={fieldState.invalid} />
              <FieldError errors={[fieldState.error]} />
            </Field>
          )}
        />
        <Controller
          name="cpf"
          control={form.control}
          render={({ field, fieldState }) => (
            <Field data-invalid={fieldState.invalid}>
              <FieldLabel htmlFor="buyer-cpf">CPF</FieldLabel>
              <Input
                {...field}
                id="buyer-cpf"
                inputMode="numeric"
                placeholder="000.000.000-00"
                aria-invalid={fieldState.invalid}
                onChange={(event) => field.onChange(formatCpfInput(event.target.value))}
              />
              <FieldError errors={[fieldState.error]} />
            </Field>
          )}
        />
        <Controller
          name="whatsapp"
          control={form.control}
          render={({ field, fieldState }) => (
            <Field data-invalid={fieldState.invalid}>
              <FieldLabel htmlFor="buyer-whatsapp">WhatsApp</FieldLabel>
              <Input
                {...field}
                id="buyer-whatsapp"
                type="tel"
                inputMode="tel"
                autoComplete="tel-national"
                placeholder="(85) 90000-0000"
                aria-invalid={fieldState.invalid}
                onChange={(event) => field.onChange(formatPhoneInput(event.target.value))}
              />
              <FieldError errors={[fieldState.error]} />
            </Field>
          )}
        />
        <Controller
          name="email"
          control={form.control}
          render={({ field, fieldState }) => (

              <FieldLabel htmlFor="buyer-email">E-mail</FieldLabel>

        <Controller
          name="acceptPrivacy"
          control={form.control}
          render={({ field, fieldState }) => (
            <Field data-invalid={fieldState.invalid} orientation="horizontal">
              <Checkbox
                id="buyer-privacy"
                checked={field.value === true}
                onCheckedChange={(checked) => field.onChange(checked === true)}
                aria-invalid={fieldState.invalid}

                </Link>
              </FieldLabel>
              <FieldError errors={[fieldState.error]} />
            </Field>
          )}
        />
        <Button type="submit" variant="cta" size="xl" disabled={isPending || items.length === 0}>

    </form>
  )
}
```

```tsx
import type { Metadata } from "next"
import { notFound } from "next/navigation"

import { ServerError } from "@/components/feedback"
import { getStoreRepository } from "@/server/repositories/store.repository"
import type { OrderSummary } from "@/types/order.interface"

import { OrderDetails } from "./components"

export const metadata: Metadata = { title: "Meu pedido", robots: { index: false } }

/** Bloqueia (sem <Suspense>) para que o notFound() rode antes do streaming e devolva 404 de verdade. */
export const instant = false

const ORDER_CODE_PATTERN = /^VTX-P-[2-9A-Z]{4}$/

export default async function OrderPage({ params }: PageProps<"/pedido/[code]">) {
  const code = decodeURIComponent((await params).code).toUpperCase()
  if (!ORDER_CODE_PATTERN.test(code)) notFound()

  let order: OrderSummary | null
  try {
    order = await getStoreRepository().getOrderSummaryByCode(code)
  } catch (error) {
    console.error("[OrderPage]", error)
    return (
      <div className="mx-auto max-w-2xl px-4 py-10">
        <ServerError title="Não foi possível carregar o pedido" />
      </div>
    )
  }
  if (!order) notFound()

  return (
    <div className="mx-auto max-w-2xl px-4 py-10">
      <OrderDetails order={order} />
    </div>
  )
}
```

Em `src/server/errors.ts`, acrescentar `ProductNotFoundError` ("Produto não encontrado") e a lista `STORE_CONFLICT_ERROR_NAMES` (+ tipo `StoreConflictErrorName` e o guard `isStoreConflictErrorName`). Os dois repositórios lançam `ProductNotFoundError` quando `saveProduct` recebe um id inexistente — o mock nunca cai na criação; o Prisma confere a existência dentro da transação e mapeia `P2025`.

- **Por que mapear pelo `name` e não com `instanceof`:** o repositório mock vive em `globalThis` e sobrevive ao HMR guardando as classes de erro antigas, então `instanceof` falha (slug duplicado virava 500). Um teste garante que `new X().name` bate com a lista.
- **Por que validar o `id`:** a action é um endpoint público; um id vazio/estranho vira 400 "Produto inválido" e um id inexistente vira 404, nunca um produto novo.

```ts
// src/app/actions/admin-products.ts
"use server"

import { revalidatePath } from "next/cache"
import { z } from "zod"

import { productFormSchema, type ProductFormInput } from "@/lib/validations/product.schema"
import { getAdminSession } from "@/server/auth/require-admin"
import { isStoreConflictErrorName, type StoreConflictErrorName } from "@/server/errors"
import { getStoreRepository } from "@/server/repositories/store.repository"
import type { HttpResponse } from "@/types/http-response.interface"

const productIdSchema = z.string().min(1).optional()

/**
 * Erros do repositório (src/server/errors.ts) → código da resposta.
 * Compara pelo `name` (fixado em cada classe) e não com `instanceof`: o repositório mock vive em
 * `globalThis` e sobrevive ao HMR guardando as classes de erro antigas.
 */
const CONFLICT_ERROR_CODES: Record<StoreConflictErrorName, string> = {
  DuplicateSlugError: "DUPLICATE_SLUG",
  DuplicateVariantLabelError: "DUPLICATE_VARIANT_LABEL",
  StockChangedError: "STOCK_CHANGED",
  VariantNotOwnedError: "VARIANT_NOT_OWNED",
}

export async function saveProductAction(input: ProductFormInput, id?: string): Promise<HttpResponse<{ id: string }>> {
  if (!(await getAdminSession())) return { status: 401, error: "UNAUTHORIZED", title: "Sessão expirada. Entre novamente." }

  const parsedId = productIdSchema.safeParse(id)
  if (!parsedId.success) return { status: 400, error: "INVALID_INPUT", title: "Produto inválido" }

  const parsed = productFormSchema.safeParse(input)
  if (!parsed.success) {
    return { status: 400, error: "INVALID_INPUT", title: parsed.error.issues[0]?.message ?? "Verifique os campos" }
  }

  const productId = parsedId.data
  try {
    const product = await getStoreRepository().saveProduct(parsed.data, productId)
    revalidatePath("/loja")
    revalidatePath(`/loja/${product.slug}`)
    revalidatePath("/admin/loja/produtos")
    revalidatePath("/")
    return {
      status: productId ? 200 : 201,
      data: { id: product.id },
      title: productId ? "Produto atualizado" : "Produto cadastrado",
    }
  } catch (error) {
    if (error instanceof Error && error.name === "ProductNotFoundError") {
      return { status: 404, error: "NOT_FOUND", title: error.message }
    }
    if (error instanceof Error && isStoreConflictErrorName(error.name)) {
      return { status: 409, error: CONFLICT_ERROR_CODES[error.name], title: error.message }
    }
    console.error("[saveProductAction]", error)
    return { status: 500, error: "INTERNAL", title: "Não foi possível salvar o produto" }
  }
}
```

```tsx
        <FieldLegend>Tamanhos e estoque</FieldLegend>
        <ul className="flex flex-col gap-3">
          {variants.fields.map((variant, index) => (
            <VariantFieldRow
              key={variant.id}
              control={form.control}
              index={index}
              canRemove={!defaultValues.variants[index]?.id && variants.fields.length > 1}
              onRemove={() => variants.remove(index)}
            />
          ))}
        </ul>
        <FieldError errors={[form.formState.errors.variants?.root, form.formState.errors.variants]} />
        <Button type="button" variant="outline" className="w-fit" onClick={() => variants.append({ label: "", stock: "0" })}>
          <Plus /> Adicionar tamanho
        </Button>
      </FieldSet>

      <FieldGroup>
        <Controller
          name="isActive"
          control={form.control}
          render={({ field }) => (
            <Field orientation="horizontal">
              <Checkbox id="product-active" checked={field.value} onCheckedChange={(checked) => field.onChange(checked === true)} />
              <FieldLabel htmlFor="product-active" className="font-normal">Ativo (visível na loja)</FieldLabel>
            </Field>
          )}
        />
        <Controller
          name="isFeatured"
          control={form.control}
          render={({ field }) => (
            <Field orientation="horizontal">
              <Checkbox id="product-featured" checked={field.value} onCheckedChange={(checked) => field.onChange(checked === true)} />
              <FieldLabel htmlFor="product-featured" className="font-normal">Destaque na home</FieldLabel>
            </Field>
          )}
        />
      </FieldGroup>

      <Button type="submit" size="xl" disabled={isPending} className="w-fit">
        {isPending ? "Salvando..." : "Salvar produto"}
      </Button>
    </form>
  )
}
```

```tsx
import type { Metadata } from "next"
import { notFound } from "next/navigation"
import { connection } from "next/server"

/**
 * A página bloqueia (sem <Suspense>) até ler o produto: assim o `notFound()` roda antes do streaming
 * começar e devolve um 404 de verdade (mesmo padrão de /loja/[slug]).
 */
export const instant = false

export const metadata: Metadata = { title: "Editar produto" }

export default async function EditProductPage({ params }: PageProps<"/admin/loja/produtos/[id]">) {
  await connection()
  const { id } = await params

  let product: StoreProduct | null
  try {
    product = await getStoreRepository().getProductById(id)
  } catch (error) {
    console.error("[EditProductPage]", error)
    return (
      <section className="flex flex-col gap-6">
        <h1 className="text-2xl font-extrabold text-brand-navy">Editar produto</h1>
        <ServerError title="Não foi possível carregar o produto" />
      </section>
    )
  }

  if (!product) notFound()

  return (
    <section className="flex flex-col gap-6">
      <EditProduct product={product} />
    </section>
  )
}
```
